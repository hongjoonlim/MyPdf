cmake_minimum_required(VERSION 3.22)
project(MyPdf VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
find_package(ZLIB REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND NOT (POPPLER_INCLUDE_DIR AND POPPLER_LIBRARY))
  pkg_check_modules(POPPLER QUIET IMPORTED_TARGET poppler-qt6)
endif()
if(TARGET PkgConfig::POPPLER)
  set(MYPDF_POPPLER_TARGET PkgConfig::POPPLER)
else()
  find_path(POPPLER_INCLUDE_DIR poppler-qt6.h PATH_SUFFIXES poppler/qt6)
  find_library(POPPLER_LIBRARY NAMES poppler-qt6)
  if(NOT POPPLER_INCLUDE_DIR OR NOT POPPLER_LIBRARY)
    message(FATAL_ERROR
      "Poppler Qt 6 development files were not found (poppler-qt6.h / libpoppler-qt6). "
      "macOS: brew install poppler-qt6 pkgconf, then use ./scripts/build.sh. "
      "Ubuntu: install libpoppler-qt6-dev. The plain poppler package is not sufficient. "
      "For a custom installation, set POPPLER_INCLUDE_DIR and POPPLER_LIBRARY explicitly.")
  endif()
  add_library(MyPdf_poppler UNKNOWN IMPORTED)
  set_target_properties(MyPdf_poppler PROPERTIES
    IMPORTED_LOCATION "${POPPLER_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${POPPLER_INCLUDE_DIR}")
  set(MYPDF_POPPLER_TARGET MyPdf_poppler)
endif()
find_program(QPDF_EXECUTABLE NAMES qpdf)
set(MYPDF_CONFIGURED_QPDF "")
if(QPDF_EXECUTABLE)
  file(TO_CMAKE_PATH "${QPDF_EXECUTABLE}" MYPDF_CONFIGURED_QPDF)
endif()
configure_file(cmake/build_config.h.in generated/build_config.h @ONLY)

add_library(MyPdf_options INTERFACE)
if(MSVC)
  target_compile_options(MyPdf_options INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(MyPdf_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
qt_add_resources(TRANSLATION_RESOURCE translations/translations.qrc)
add_library(pdfcore src/docx.cpp src/document.cpp src/i18n.cpp src/toolpaths.cpp ${TRANSLATION_RESOURCE})
target_include_directories(pdfcore PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(pdfcore PUBLIC Qt6::Widgets ZLIB::ZLIB ${MYPDF_POPPLER_TARGET} MyPdf_options)

qt_add_resources(APP_ICON_RESOURCE assets/icons/app.qrc)
qt_add_executable(MyPdf WIN32 MACOSX_BUNDLE src/main.cpp src/icons.cpp ${APP_ICON_RESOURCE})
target_link_libraries(MyPdf PRIVATE pdfcore)
if(WIN32)
  enable_language(RC)
  configure_file(packaging/windows.rc.in generated/windows.rc @ONLY)
  target_sources(MyPdf PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/windows.rc")
elseif(APPLE)
  target_sources(MyPdf PRIVATE assets/icons/MyPdf.icns)
  set_source_files_properties(assets/icons/MyPdf.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(MyPdf PROPERTIES
    OUTPUT_NAME MyPdf
    MACOSX_BUNDLE_BUNDLE_NAME MyPdf
    MACOSX_BUNDLE_GUI_IDENTIFIER org.MyPdf.MyPdf
    MACOSX_BUNDLE_ICON_FILE MyPdf.icns
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}")
endif()

include(CTest)
if(BUILD_TESTING)
  if(NOT QPDF_EXECUTABLE)
    message(FATAL_ERROR "Tests require qpdf on PATH or -DQPDF_EXECUTABLE=/path/to/qpdf")
  endif()
  add_executable(docx_test tests/docx_test.cpp src/icons.cpp)
  target_link_libraries(docx_test PRIVATE pdfcore)
  add_test(NAME native_docx COMMAND docx_test)
  add_executable(toolpaths_test tests/toolpaths_test.cpp)
  target_link_libraries(toolpaths_test PRIVATE pdfcore)
  add_test(NAME tool_discovery COMMAND toolpaths_test)
  add_executable(pdfcore_test tests/document_test.cpp)
  target_link_libraries(pdfcore_test PRIVATE pdfcore)
  add_test(NAME pdf_operations COMMAND pdfcore_test)
  add_executable(localization_test tests/localization_test.cpp src/icons.cpp)
  target_link_libraries(localization_test PRIVATE pdfcore)
  add_test(NAME localization COMMAND localization_test)
  add_executable(signature_test tests/signature_test.cpp src/icons.cpp)
  target_link_libraries(signature_test PRIVATE pdfcore)
  add_test(NAME signature_cancellation COMMAND signature_test)
  add_executable(objects_test tests/objects_test.cpp src/icons.cpp)
  target_link_libraries(objects_test PRIVATE pdfcore)
  add_test(NAME editable_objects COMMAND objects_test)
  add_executable(object_edit_test tests/object_edit_test.cpp src/icons.cpp)
  target_link_libraries(object_edit_test PRIVATE pdfcore)
  add_test(NAME object_editing COMMAND object_edit_test)
  set_tests_properties(object_editing editable_objects native_docx pdf_operations localization signature_cancellation PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;MYPDF_QPDF=${QPDF_EXECUTABLE}")
endif()

include(GNUInstallDirs)
install(TARGETS MyPdf BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
  install(FILES assets/icons/MyPdf.png DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps)
  install(FILES assets/icons/MyPdf.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
  configure_file(cmake/install_desktop.cmake.in generated/install_desktop.cmake @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/generated/install_desktop.cmake")
endif()
